# Install: bash scripts/install_crontab.sh
# keepalive: every 2 min — tunnel recycle, Docker/Prometheus, gunicorn, metrics_exporter, cold-start (backup if anything dies).
# For resilient SSH to Kafka: bash scripts/install_autossh.sh (once per VM).

PROJECT_ROOT=__PROJECT_ROOT__

# Full stack recovery after reboot (tunnel + services; Docker needs sudo or docker group)
@reboot sleep 120 && cd $PROJECT_ROOT && bash scripts/reboot_recover.sh >> logs/reboot_recover.log 2>&1

# Frequent health pass (primary backup for disconnects / dead exporter)
*/2 * * * * cd $PROJECT_ROOT && bash scripts/keepalive.sh >> logs/keepalive.log 2>&1

# Extra tunnel attempt every 5 min (idempotent; complements keepalive)
*/5 * * * * cd $PROJECT_ROOT && bash scripts/ensure_kafka_tunnel.sh >> logs/tunnel.log 2>&1

# API health: if /health fails, restart gunicorn (belt-and-suspenders with keepalive)
*/5 * * * * curl -sf http://localhost:8082/health > /dev/null || (cd $PROJECT_ROOT && bash scripts/restart_service.sh >> logs/cron_restart.log 2>&1)

# Retrain every 3 days 02:00 UTC
0 2 */3 * * cd $PROJECT_ROOT && bash scripts/retrain_and_deploy.sh --skip-pull >> logs/cron_retrain.log 2>&1

# Cold-start oneshot every 30 min (5 min window; complements continuous daemon from keepalive)
*/30 * * * * cd $PROJECT_ROOT && bash scripts/ensure_kafka_tunnel.sh >> logs/tunnel.log 2>&1 && python3 scripts/update_cold_start.py --mode oneshot --duration 300 >> logs/cron_cold_start.log 2>&1
